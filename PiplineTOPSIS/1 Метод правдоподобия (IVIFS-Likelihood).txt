АЛГОРИТМ: IVIFS-Likelihood (TOPSISort-L подход)
ВХОД: Взвешенная IVIFS матрица W[n][m], критерии C[m]
ВЫХОД: Матрица решений ζ[n][m]

ФУНКЦИЯ ВЫПОЛНИТЬ_IVIFS_LIKELIHOOD(W, C):
    n ← РАЗМЕР(W)      // количество альтернатив
    m ← РАЗМЕР(W[0])   // количество критериев
    ζ ← НОВЫЙ_МАССИВ[n][m] // матрица правдоподобий
    
    ДЛЯ каждого критерия j от 0 ДО m-1:
        ДЛЯ каждой альтернативы i от 0 ДО n-1:
            sum_likelihood ← 0
            count ← 0
            
            // Попарное сравнение с остальными альтернативами
            ДЛЯ каждой альтернативы k от 0 ДО n-1, ГДЕ k ≠ i:
                // Вычисление правдоподобия L(W[i][j] ≥ W[k][j])
                likelihood ← ВЫЧИСЛИТЬ_LIKELIHOOD(W[i][j], W[k][j])
                
                // Учёт типа критерия
                ЕСЛИ C[j].type == 'positive':
                    sum_likelihood ← sum_likelihood + likelihood
                ИНАЧЕ: // negative criterion
                    sum_likelihood ← sum_likelihood + (1 - likelihood)
                
                count ← count + 1
            
            // Нормализация
            ЕСЛИ count > 0:
                ζ[i][j] ← sum_likelihood / count
            ИНАЧЕ:
                ζ[i][j] ← 0
    
    ВЕРНУТЬ ζ
КОНЕЦ_ФУНКЦИИ

ФУНКЦИЯ ВЫЧИСЛИТЬ_LIKELIHOOD(ivifs1, ivifs2):
    // ivifs = [[τ⁻, τ⁺], [υ⁻, υ⁺]]
    τ1_min, τ1_max ← ivifs1[0]
    υ1_min, υ1_max ← ivifs1[1]
    τ2_min, τ2_max ← ivifs2[0]
    υ2_min, υ2_max ← ivifs2[1]
    
    // Нижняя вероятность L⁻
    числитель_ниж ← (1 - υ2_min) - τ1_min
    знаменатель_ниж ← (1 - τ1_min - υ1_max) + (1 - τ2_max - υ2_min)
    ЕСЛИ знаменатель_ниж ≠ 0:
        L_lower ← МАКС(1 - МАКС(числитель_ниж/знаменатель_ниж, 0), 0)
    ИНАЧЕ:
        L_lower ← 0
    
    // Верхняя вероятность L⁺
    числитель_верх ← (1 - υ2_max) - τ1_max
    знаменатель_верх ← (1 - τ1_max - υ1_min) + (1 - τ2_min - υ2_max)
    ЕСЛИ знаменатель_верх ≠ 0:
        L_upper ← МАКС(1 - МАКС(числитель_верх/знаменатель_верх, 0), 0)
    ИНАЧЕ:
        L_upper ← 0
    
    // Общая вероятность
    ВЕРНУТЬ 0.5 * (L_lower + L_upper)
КОНЕЦ_ФУНКЦИИ
